{
  ...
}:
{
  flake.nixosModules.emacs = { pkgs, ... }:
  let
    myEmacs = pkgs.emacs-pgtk.pkgs.withPackages (pkgs: [
      pkgs.evil
      pkgs.evil-collection
      pkgs.doom-themes
      pkgs.nix-mode
    ]);
    emacsInit = pkgs.writeText "emacs-init.el" ''
      ;; -*- lexical-binding: t; -*-
      (require 'evil)
      (evil-mode 1)
      (load-theme 'doom-one t)
    '';
  in
  {
    environment.systemPackages = [
      myEmacs
    ];
    systemd.tmpfiles.rules = [
      "d /home/ty/.config/emacs 0755 ty users -"
      "C+ /home/ty/.config/emacs/init.el 0644 ty users - ${emacsInit}"
    ];
  };
}
